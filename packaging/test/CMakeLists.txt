cmake_minimum_required(VERSION 3.15)
project(tinyusd_package_test LANGUAGES C CXX)

set(TINYUSD_TESTS_DIR "" CACHE PATH "The repository's tests folder, to run its C and C++ suites against the package")
option(TINYUSD_TEST_WRITE "Run the write API tests (turn off for packages built without the write feature)" ON)

find_package(tinyusd CONFIG REQUIRED)

enable_testing()

add_executable(package_test package_test.cpp)
target_link_libraries(package_test PRIVATE tinyusd::tinyusd)
target_compile_features(package_test PRIVATE cxx_std_17)
add_test(NAME package COMMAND package_test)

if(TINYUSD_TESTS_DIR)
  set(_src "${TINYUSD_TESTS_DIR}/cpp")
  set(_data "${TINYUSD_TESTS_DIR}/data")
  set(_suites query)
  if(TINYUSD_TEST_WRITE)
    list(APPEND _suites write)
    set(_out "${CMAKE_CURRENT_BINARY_DIR}/write_out")
    file(MAKE_DIRECTORY "${_out}")
  endif()
  foreach(suite ${_suites})
    add_executable(test_${suite}_cpp "${_src}/test_${suite}.cpp")
    target_link_libraries(test_${suite}_cpp PRIVATE tinyusd::tinyusd)
    target_compile_features(test_${suite}_cpp PRIVATE cxx_std_17)
    add_executable(test_${suite}_c "${_src}/test_${suite}.c")
    target_link_libraries(test_${suite}_c PRIVATE tinyusd::tinyusd)
    set_target_properties(test_${suite}_c PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
    set(_args "${_data}")
    if(suite STREQUAL "write")
      list(APPEND _args "${_out}")
    endif()
    add_test(NAME cpp_${suite} COMMAND test_${suite}_cpp ${_args})
    add_test(NAME c_${suite} COMMAND test_${suite}_c ${_args})
  endforeach()
endif()
